cmake_minimum_required(VERSION 3.21)
project(HZLFluidDynamics VERSION 0.1.0 LANGUAGES CXX)
option(HZL_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(HZL_CUDA "Build the CUDA solver and comparison benchmark" OFF)
option(HZL_MPI "Build the distributed-memory ensemble executable" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(hzl src/fft.cpp src/solver.cpp src/inference.cpp src/adjoint.cpp src/filter.cpp src/reduced.cpp src/monte_carlo.cpp src/surrogate.cpp)
target_include_directories(hzl PUBLIC include)
target_link_libraries(hzl PUBLIC Threads::Threads)
target_compile_options(hzl PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
if(HZL_SANITIZERS)
  target_compile_options(hzl PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(hzl PUBLIC -fsanitize=address,undefined)
endif()
add_executable(hzl-flow src/main.cpp)
target_link_libraries(hzl-flow PRIVATE hzl)
if(HZL_CUDA)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "CUDA architectures")
  endif()
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  add_library(hzl-cuda src/cuda_solver.cu)
  target_include_directories(hzl-cuda PUBLIC include)
  target_link_libraries(hzl-cuda PUBLIC hzl CUDA::cudart CUDA::cufft)
  set_target_properties(hzl-cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
  add_executable(hzl-cuda-benchmark src/cuda_benchmark.cpp)
  target_link_libraries(hzl-cuda-benchmark PRIVATE hzl-cuda)
  add_executable(hzl-cuda-ensemble-benchmark src/cuda_ensemble_benchmark.cpp)
  target_link_libraries(hzl-cuda-ensemble-benchmark PRIVATE hzl-cuda)
endif()
if(HZL_MPI)
  find_package(MPI REQUIRED COMPONENTS CXX)
  add_executable(hzl-mpi-ensemble src/mpi_ensemble.cpp)
  target_link_libraries(hzl-mpi-ensemble PRIVATE hzl MPI::MPI_CXX)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(hzl-verify tests/verification.cpp)
  target_link_libraries(hzl-verify PRIVATE hzl)
  add_test(NAME numerical_statistical_verification COMMAND hzl-verify)
endif()
